\newcommand{\PosterText}[8][left]{%
  \pgfmathsetlengthmacro{\PosterTextWidth}{(#4)*1bp}%
  \node[anchor=north west,inner sep=0pt,outer sep=0pt,
    text width=\PosterTextWidth,align=#1,text=#7,
    font=\sffamily\fontsize{#5bp}{#6bp}\selectfont]
    at ({#2},{#3}) {\strut #8};%
}
% \BulletText{x}{top}{width}{font size}{line spacing}{text}
\newcommand{\BulletText}[6]{%
  \fill[Teal] ({#1+3},{#2+8}) circle[radius=2bp];
  \PosterText{#1+14}{#2}{#3-14}{#4}{#5}{Ink}{#6}%
}
% \RoundBox{x}{top}{width}{height}{radius}{fill colour}{border colour}
\newcommand{\RoundBox}[7]{%
  \draw[fill=#6,draw=#7,line width=.8bp,rounded corners=#5bp]
    (#1,#2) rectangle ({#1+#3},{#2+#4});%
}
% \Lane{x}{width}{accent}{background}{heading}
\newcommand{\Lane}[5]{%
  \RoundBox{#1}{122}{#2}{565}{12}{#4}{LaneBorder}
  \path[fill=#3,rounded corners=10bp]
    (#1,122) rectangle ({#1+#2},155);
  \PosterText[center]{#1+8}{130}{#2-16}{15.2}{19.5}{white}{\textbf{#5}}%
}
% \StepCard{number}{title}{x}{top}{width}{height}{accent}
\newcommand{\StepCard}[7]{%
  \RoundBox{#3}{#4}{#5}{#6}{10}{white}{CardBorder}
  \path[fill=#7,rounded corners=2bp]
    ({#3+1},{#4+1}) rectangle ({#3+#5-1},{#4+8});
  \fill[#7] ({#3+29},{#4+33}) circle[radius=17bp];
  \node[anchor=center,inner sep=0pt,text=white,
    font=\sffamily\bfseries\fontsize{21bp}{26bp}\selectfont]
    at ({#3+29},{#4+33}) {#1};
  \PosterText{#3+56}{#4+21}{#5-68}{15.8}{20.2}{Ink}{\textbf{#2}}%
}
